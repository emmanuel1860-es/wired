# Formato: abreviatura = significado (aleman estandar, misma funcion gramatical). "x = -" borra una entrada integrada.
# Las claves tambien suman 1 punto de "es aleman" en la deteccion: por eso se evitan las que son palabras corrientes

# ---------- 0. Correcciones de las entradas integradas (auditoria) ----------
# Cada clave tambien suma 1 punto de "es aleman" en la deteccion; por eso no se anaden palabras que otros idiomas usan igual
# (is net ok u oda wat dat des dos nem hats grad ma trade...) y aqui se borran las que ya estaban.
# (Los comentarios van siempre en linea propia: el lector no admite comentarios al final de una entrada.)
# alter: adjetivo ("ein alter Thron") y sustantivo ("dein Alter"); "Freund" rompia ambos. Google ya traduce bien "Alter," como vocativo.
alter = -
# is: la palabra mas corriente del ingles y del neerlandes; contaminaba la deteccion de esos idiomas.
is = -
# net: neerlandes (= justo) e ingles; en aleman solo es dialecto del sur.
net = -
# ne: es "nein" (ne, danke) o "eine" (ne Lampe); sin contexto no se puede expandir y Google lo entiende bien.
ne = -
# u: "you" en ingles.
u = -
# ok: no cambia nada y suma puntos de aleman a cualquier idioma.
ok = -
# wg: "wie geht's" o "wegen", ambiguo.
wg = -
# bb: era "bis bald"; en el chat es "bye bye".
bb = tschüss
# mega: era "sehr" y "mega Raum" salia "sehr Raum"; "super" sirve delante de nombres, adjetivos y suelto.
mega = super
# digga: era "Freund".
digga = Kumpel
